set(M88 ${CMAKE_CURRENT_LIST_DIR}/../../third_party/m88/src)
file(GLOB M88_COMMON ${M88}/common/*.cpp)
file(GLOB M88_DEVICES ${M88}/devices/*.cpp)
list(FILTER M88_DEVICES EXCLUDE REGEX "Z80(_x86|Debug|Test|diag)\\.cpp$")
file(GLOB M88_MACHINE ${M88}/pc88/*.cpp)
list(FILTER M88_MACHINE EXCLUDE REGEX "(ioview|memview)\\.cpp$")
add_library(pc88 SHARED ${M88_COMMON} ${M88_DEVICES} ${M88_MACHINE} platform.cpp adapter.cpp)
target_include_directories(pc88 PRIVATE platform ${M88} ${M88}/common ${M88}/devices ${M88}/pc88)
target_compile_features(pc88 PRIVATE cxx_std_17)
target_compile_options(pc88 PRIVATE -fno-strict-aliasing -Wno-invalid-source-encoding -Wno-writable-strings)
target_link_libraries(pc88 PRIVATE z log)
set_target_properties(pc88 PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
